// ============================================================
// Favorites State
// ============================================================

let favorites = [];

// ============================================================
// Toggle a product in/out of favorites
// ============================================================

function toggleFavorite(productId) {
  if (!products[productId]) {
    console.warn(`Product ID "${productId}" not found in catalog.`);
    return;
  }

  const btn = document.querySelector(`[data-product-id="${productId}"]`);
  const alreadySaved = favorites.some(item => item.id === productId);

  if (alreadySaved) {
    favorites = favorites.filter(item => item.id !== productId);
    if (btn) {
      btn.textContent = "Save to Favorites";
      btn.classList.remove("favorited");
      btn.setAttribute("aria-pressed", "false");
    }
  } else {
    favorites.push({ id: productId, ...products[productId] });
    if (btn) {
      btn.textContent = "♥ Saved";
      btn.classList.add("favorited");
      btn.setAttribute("aria-pressed", "true");
    }
  }

  saveFavorites();
  renderFavorites();
}

// ============================================================
// Persist favorites to localStorage
// ============================================================

function saveFavorites() {
  try {
    localStorage.setItem("bakeryFavorites", JSON.stringify(favorites));
  } catch (e) {
    console.error("Could not save favorites to localStorage:", e);
  }
}

// ============================================================
// Render favorites list grouped by category
// ============================================================

function renderFavorites() {
  const list = document.getElementById("favorites-list");
  if (!list) return;

  list.innerHTML = "";

  if (favorites.length === 0) {
    list.innerHTML = "<p>No favorites saved yet.</p>";
    return;
  }

  // Group favorites by category
  const grouped = favorites.reduce((acc, item) => {
    const cat = item.category || "Other";
    if (!acc[cat]) acc[cat] = [];
    acc[cat].push(item);
    return acc;
  }, {});

  // Render each category group
  Object.entries(grouped).forEach(([category, items]) => {
    const section = document.createElement("div");
    section.classList.add("favorites-category");

    const heading = document.createElement("h3");
    heading.textContent = category;
    section.appendChild(heading);

    const ul = document.createElement("ul");
    items.forEach(item => {
      const li = document.createElement("li");
      li.innerHTML = `
        <span class="fav-name">${item.name}</span>
        <span class="fav-price">${item.price}</span>
        <button
          class="remove-favorite"
          data-product-id="${item.id}"
          aria-label="Remove ${item.name} from favorites"
          onclick="toggleFavorite('${item.id}')">
          ✕
        </button>
      `;
      ul.appendChild(li);
    });

    section.appendChild(ul);
    list.appendChild(section);
  });
}

// ============================================================
// Load favorites from localStorage on page load
// ============================================================

function loadFavorites() {
  try {
    const stored = localStorage.getItem("bakeryFavorites");
    if (stored) {
      const parsed = JSON.parse(stored);

      // Validate stored IDs still exist in the current catalog
      favorites = parsed.filter(item => products[item.id]);

      renderFavorites();

      // Restore button states for all favorited products
      favorites.forEach(item => {
        const btn = document.querySelector(`[data-product-id="${item.id}"]`);
        if (btn) {
          btn.textContent = "♥ Saved";
          btn.classList.add("favorited");
          btn.setAttribute("aria-pressed", "true");
        }
      });
    }
  } catch (e) {
    console.error("Could not load favorites from localStorage:", e);
    favorites = [];
  }
}

// ============================================================
// Clear all favorites
// ============================================================

function clearAllFavorites() {
  favorites = [];
  saveFavorites();
  renderFavorites();

  // Reset all buttons on the page
  document.querySelectorAll("[data-product-id]").forEach(btn => {
    btn.textContent = "Save to Favorites";
    btn.classList.remove("favorited");
    btn.setAttribute("aria-pressed", "false");
  });
}

// ============================================================
// Init
// ============================================================

window.addEventListener("DOMContentLoaded", loadFavorites);
