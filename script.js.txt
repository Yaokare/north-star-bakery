var FAVORITES_KEY = "nsb_favorites";
var NAME_KEY      = "nsb_contact_name";

var products = [
  { id: "classic-sourdough",   name: "Classic Sourdough",                category: "Breads"   },
  { id: "honey-oat-loaf",      name: "Honey Oat Loaf",                   category: "Breads"   },
  { id: "seeded-rye",          name: "Seeded Rye",                       category: "Breads"   },
  { id: "rosemary-focaccia",   name: "Rosemary Focaccia",                category: "Breads"   },
  { id: "butter-croissant",    name: "Butter Croissant",                 category: "Pastries" },
  { id: "fruit-danish",        name: "Seasonal Fruit Danish",            category: "Pastries" },
  { id: "morning-bun",         name: "Cinnamon Morning Bun",             category: "Pastries" },
  { id: "choc-almond",         name: "Chocolate Almond Croissant",       category: "Pastries" },
  { id: "vanilla-cake",        name: "Classic Vanilla Celebration Cake", category: "Cakes"    },
  { id: "choc-ganache-cake",   name: "Dark Chocolate Ganache Cake",      category: "Cakes"    },
  { id: "lemon-drizzle",       name: "Lemon Drizzle Loaf",               category: "Cakes"    },
  { id: "carrot-cake",         name: "Carrot Cake",                      category: "Cakes"    }
];

/* ----------------------------------------------------------
   STORAGE HELPERS
   ---------------------------------------------------------- */
function loadFavorites() {
  try {
    var stored = localStorage.getItem(FAVORITES_KEY);
    var result = stored ? JSON.parse(stored) : [];
    console.log("[NSB] loadFavorites:", result);
    return result;
  } catch (e) {
    console.error("[NSB] loadFavorites error:", e);
    return [];
  }
}

function saveFavorites(arr) {
  try {
    localStorage.setItem(FAVORITES_KEY, JSON.stringify(arr));
    console.log("[NSB] saveFavorites:", arr);
  } catch (e) {
    console.error("[NSB] saveFavorites error:", e);
  }
}

/* ----------------------------------------------------------
   TOGGLE
   ---------------------------------------------------------- */
function toggleFavorite(productId) {
  console.log("[NSB] toggleFavorite called with:", productId);
  var favorites = loadFavorites();
  var index = favorites.indexOf(productId);

  if (index === -1) {
    favorites.push(productId);
    console.log("[NSB] Added:", productId);
  } else {
    favorites.splice(index, 1);
    console.log("[NSB] Removed:", productId);
  }

  saveFavorites(favorites);
  updateFavoritesPanel();
  updateHeartButtons();
}

/* ----------------------------------------------------------
   UPDATE HEART BUTTONS
   ---------------------------------------------------------- */
function updateHeartButtons() {
  var favorites = loadFavorites();
  var buttons = document.querySelectorAll(".fav-btn");
  console.log("[NSB] updateHeartButtons — buttons found:", buttons.length);

  buttons.forEach(function (btn) {
    var productId = btn.getAttribute("data-product-id");
    if (favorites.indexOf(productId) !== -1) {
      btn.textContent = "♥ Saved";
      btn.classList.add("fav-btn--active");
      btn.setAttribute("aria-pressed", "true");
    } else {
      btn.textContent = "♡ Save";
      btn.classList.remove("fav-btn--active");
      btn.setAttribute("aria-pressed", "false");
    }
  });
}

/* ----------------------------------------------------------
   UPDATE FAVORITES PANEL
   ---------------------------------------------------------- */
function updateFavoritesPanel() {
  var panel = document.getElementById("favorites-panel");
  if (!panel) {
    console.log("[NSB] updateFavoritesPanel — panel not found, skipping");
    return;
  }

  var favorites = loadFavorites();
  var list     = document.getElementById("favorites-list");
  var emptyMsg = document.getElementById("favorites-empty");
  var countEl  = document.getElementById("favorites-count");

  if (!list || !emptyMsg || !countEl) {
    console.error("[NSB] updateFavoritesPanel — missing panel child elements");
    return;
  }

  list.innerHTML = "";

  if (favorites.length === 0) {
    emptyMsg.style.display = "block";
    countEl.textContent = "0";
    console.log("[NSB] Panel updated — empty state");
    return;
  }

  emptyMsg.style.display = "none";
  countEl.textContent = String(favorites.length);

  favorites.forEach(function (productId) {
    var product = null;
    for (var i = 0; i < products.length; i++) {
      if (products[i].id === productId) {
        product = products[i];
        break;
      }
    }

    if (!product) {
      console.warn("[NSB] No product found for id:", productId);
      return;
    }

    var li = document.createElement("li");
    li.textContent = product.name + " (" + product.category + ")";

    var removeBtn = document.createElement("button");
    removeBtn.textContent = "Remove";
    removeBtn.className = "fav-remove-btn";
    removeBtn.setAttribute("aria-label", "Remove " + product.name + " from favorites");

    removeBtn.addEventListener("click", function () {
      toggleFavorite(productId);
    });

    li.appendChild(removeBtn);
    list.appendChild(li);
  });

  console.log("[NSB] Panel updated —", favorites.length, "item(s)");
}

/* ----------------------------------------------------------
   INIT FAVORITES
   ---------------------------------------------------------- */
function initFavoritesFeature() {
  var favButtons = document.querySelectorAll(".fav-btn");
  console.log("[NSB] initFavoritesFeature — buttons found:", favButtons.length);

  if (favButtons.length === 0) {
    console.log("[NSB] No fav buttons on this page, skipping");
    return;
  }

  favButtons.forEach(function (btn) {
    btn.addEventListener("click", function (event) {
      event.preventDefault();
      var productId = btn.getAttribute("data-product-id");
      console.log("[NSB] Button clicked, productId:", productId);
      toggleFavorite(productId);
    });
  });

  updateFavoritesPanel();
  updateHeartButtons();
  console.log("[NSB] Favorites feature initialized");
}

/* ----------------------------------------------------------
   CONTACT FORM VALIDATION
   ---------------------------------------------------------- */
var validationRules = {
  name:        { minLength: 2,  errorMsg: "Please enter your full name (at least 2 characters)." },
  email:       { pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/, errorMsg: "Please enter a valid email address." },
  pickupDate:  { errorMsg: "Please select a pickup date at least 24 hours from today." },
  requestType: { errorMsg: "Please select a request type." },
  itemDetails: { minLength: 10, errorMsg: "Please describe your order in at least 10 characters." }
};

function showError(fieldId, message) {
  var field = document.getElementById(fieldId);
  if (!field) return;

  var errorEl = document.getElementById(fieldId + "-error");
  if (!errorEl) {
    errorEl = document.createElement("span");
    errorEl.id = fieldId + "-error";
    errorEl.className = "field-error";
    errorEl.setAttribute("role", "alert");
    field.parentNode.insertBefore(errorEl, field.nextSibling);
  }

  errorEl.textContent = message;
  field.classList.add("input-invalid");
  field.setAttribute("aria-describedby", fieldId + "-error");
}

function clearError(fieldId) {
  var field = document.getElementById(fieldId);
  if (!field) return;

  var errorEl = document.getElementById(fieldId + "-error");
  if (errorEl) errorEl.textContent = "";
  field.classList.remove("input-invalid");
}

function validateName() {
  var field = document.getElementById("name");
  if (!field) return true;
  if (field.value.trim().length < validationRules.name.minLength) {
    showError("name", validationRules.name.errorMsg);
    return false;
  }
  clearError("name");
  return true;
}

function validateEmail() {
  var field = document.getElementById("email");
  if (!field) return true;
  if (!validationRules.email.pattern.test(field.value.trim())) {
    showError("email", validationRules.email.errorMsg);
    return false;
  }
  clearError("email");
  return true;
}

function validatePickupDate() {
  var field            = document.getElementById("pickup-date");
  var requestTypeField = document.getElementById("request-type");
  if (!field) return true;

  var selectedDate = new Date(field.value);
  var today        = new Date();
  today.setHours(0, 0, 0, 0);

  var minHours = (requestTypeField && requestTypeField.value === "pre-order-cake") ? 48 : 24;
  var minDate  = new Date(today.getTime() + minHours * 60 * 60 * 1000);

  if (!field.value || selectedDate < minDate) {
    var msg = minHours === 48
      ? "Custom cake orders require at least 48 hours advance notice."
      : validationRules.pickupDate.errorMsg;
    showError("pickup-date", msg);
    return false;
  }
  clearError("pickup-date");
  return true;
}

function validateRequestType() {
  var field = document.getElementById("request-type");
  if (!field) return true;
  if (!field.value) {
    showError("request-type", validationRules.requestType.errorMsg);
    return false;
  }
  clearError("request-type");
  return true;
}

function validateItemDetails() {
  var field = document.getElementById("item-details");
  if (!field) return true;
  if (field.value.trim().length < validationRules.itemDetails.minLength) {
    showError("item-details", validationRules.itemDetails.errorMsg);
    return false;
  }
  clearError("item-details");
  return true;
}

function validateForm() {
  var results = [
    validateName(),
    validateEmail(),
    validatePickupDate(),
    validateRequestType(),
    validateItemDetails()
  ];

  var allValid = true;
  for (var i = 0; i < results.length; i++) {
    if (results[i] === false) {
      allValid = false;
      break;
    }
  }

  if (!allValid) {
    var firstInvalid = document.querySelector(".input-invalid");
    if (firstInvalid) firstInvalid.focus();
  }

  return allValid;
}

function saveContactName() {
  var nameField = document.getElementById("name");
  if (nameField && nameField.value.trim().length > 0) {
    localStorage.setItem(NAME_KEY, nameField.value.trim());
  }
}

function restoreContactName() {
  var nameField = document.getElementById("name");
  if (!nameField) return;
  var savedName = localStorage.getItem(NAME_KEY);
  if (savedName) nameField.value = savedName;
}

function showSuccessMessage() {
  var formSection = document.querySelector(".contact-form-section");
  if (!formSection) return;

  formSection.innerHTML =
    "<div class='form-success'>" +
    "<h2>Thank You!</h2>" +
    "<p>Your request has been received. We will get back to you within one business day.</p>" +
    "<p>We look forward to baking for you!</p>" +
    "</div>";
}

function initContactForm() {
  var form = document.querySelector(".contact-form");
  if (!form) {
    console.log("[NSB] No contact form on this page, skipping");
    return;
  }

  console.log("[NSB] initContactForm — form found");
  restoreContactName();

  var nameField        = document.getElementById("name");
  var emailField       = document.getElementById("email");
  var pickupDateField  = document.getElementById("pickup-date");
  var requestTypeField = document.getElementById("request-type");
  var itemDetailsField = document.getElementById("item-details");

  if (nameField)        nameField.addEventListener("input", validateName);
  if (emailField)       emailField.addEventListener("input", validateEmail);
  if (pickupDateField)  pickupDateField.addEventListener("change", validatePickupDate);
  if (requestTypeField) requestTypeField.addEventListener("change", function () {
    validateRequestType();
    validatePickupDate();
  });
  if (itemDetailsField) itemDetailsField.addEventListener("input", validateItemDetails);

  form.addEventListener("submit", function (event) {
    event.preventDefault();
    console.log("[NSB] Form submitted");
    if (validateForm()) {
      saveContactName();
      showSuccessMessage();
    }
  });

  console.log("[NSB] Contact form initialized");
}

/* ----------------------------------------------------------
   INIT
   ---------------------------------------------------------- */
document.addEventListener("DOMContentLoaded", function () {
  console.log("[NSB] DOMContentLoaded fired");
  initFavoritesFeature();
  initContactForm();
});
