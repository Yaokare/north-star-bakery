  <script>
    const products = {
      "classic-sourdough":              { name: "Classic Sourdough",                    price: "$9.00 – $12.00" },
      "honey-oat-loaf":                 { name: "Honey Oat Loaf",                       price: "$8.00 – $10.00" },
      "seeded-rye":                     { name: "Seeded Rye",                           price: "$9.00 – $11.00" },
      "rosemary-focaccia":              { name: "Rosemary Focaccia",                    price: "$7.00 – $14.00" },
      "butter-croissant":               { name: "Butter Croissant",                     price: "$3.50 – $4.50"  },
      "seasonal-fruit-danish":          { name: "Seasonal Fruit Danish",                price: "$4.00 – $5.00"  },
      "cinnamon-morning-bun":           { name: "Cinnamon Morning Bun",                 price: "$4.00 – $5.00"  },
      "chocolate-almond-croissant":     { name: "Chocolate Almond Croissant",           price: "$5.00 – $6.00"  },
      "classic-vanilla-celebration-cake": { name: "Classic Vanilla Celebration Cake",  price: "$45.00 – $85.00"},
      "dark-chocolate-ganache-cake":    { name: "Dark Chocolate Ganache Cake",          price: "$50.00 – $95.00"},
      "lemon-drizzle-loaf":             { name: "Lemon Drizzle Loaf",                   price: "$18.00 – $24.00"},
      "carrot-cake-cream-cheese":       { name: "Carrot Cake with Cream Cheese Frosting", price: "$40.00 – $75.00"}
    };

    let favorites = [];

    // ── Toggle ♥ / ☆ ──
    function toggleFavorite(productId) {
      if (!products[productId]) {
        console.warn(`Product ID "${productId}" not found.`);
        return;
      }

      const btn = document.querySelector(`[data-product-id="${productId}"]`);
      const alreadySaved = favorites.some(item => item.id === productId);

      if (alreadySaved) {
        favorites = favorites.filter(item => item.id !== productId);
        if (btn) {
          btn.textContent = "☆ Save to Favorites";
          btn.classList.remove("favorited");
          btn.setAttribute("aria-label", `Add ${products[productId].name} to favorites`);
        }
      } else {
        favorites.push({ id: productId, ...products[productId] });
        if (btn) {
          btn.textContent = "♥ Saved";
          btn.classList.add("favorited");
          btn.setAttribute("aria-label", `Remove ${products[productId].name} from favorites`);
        }
      }

      saveFavorites();
      renderFavorites();
    }

    // ── Save to localStorage ──
    function saveFavorites() {
      try {
        localStorage.setItem("bakeryFavorites", JSON.stringify(favorites));
      } catch (e) {
        console.error("Could not save favorites:", e);
      }
    }

    // ── Render favorites list ──
    function renderFavorites() {
      const list = document.getElementById("favorites-list");
      if (!list) return;

      list.innerHTML = "";

      if (favorites.length === 0) {
        list.innerHTML = "<p>No favorites saved yet.</p>";
        return;
      }

      favorites.forEach(item => {
        const li = document.createElement("li");
        li.textContent = `${item.name} — ${item.price}`;
        list.appendChild(li);
      });
    }

    // ── Load from localStorage ──
    function loadFavorites() {
      try {
        const stored = localStorage.getItem("bakeryFavorites");
        if (stored) {
          const parsed = JSON.parse(stored);
          favorites = parsed.filter(item => products[item.id]);

          favorites.forEach(item => {
            const btn = document.querySelector(`[data-product-id="${item.id}"]`);
            if (btn) {
              btn.textContent = "♥ Saved";
              btn.classList.add("favorited");
              btn.setAttribute("aria-label", `Remove ${item.name} from favorites`);
            }
          });
        }
      } catch (e) {
        console.error("Could not load favorites:", e);
        favorites = [];
      }

      renderFavorites();
    }

    window.addEventListener("DOMContentLoaded", loadFavorites);
  </script>
