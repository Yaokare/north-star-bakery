/* ----------------------------------------------------------
   DATA: Validation rules object
   ---------------------------------------------------------- */
const validationRules = {
  name:        { minLength: 2,  errorMsg: "Please enter your full name (at least 2 characters)." },
  email:       { pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/, errorMsg: "Please enter a valid email address." },
  pickupDate:  { errorMsg: "Please select a pickup date at least 24 hours from today." },
  requestType: { errorMsg: "Please select a request type." },
  itemDetails: { minLength: 10, errorMsg: "Please describe your order in at least 10 characters." }
};

/* ----------------------------------------------------------
   STORAGE KEYS
   ---------------------------------------------------------- */
const FAVORITES_KEY = "nsb_favorites";
const NAME_KEY      = "nsb_contact_name";

/* ============================================================
   FEATURE 1 — PRODUCT FAVORITES TRACKER
   Appears on products.html
   ============================================================ */

/**
 * loadFavorites()
 * Reads the favorites array from localStorage.
 * Returns an empty array if nothing is stored yet.
 */
function loadFavorites() {
  const stored = localStorage.getItem(FAVORITES_KEY);
  return stored ? JSON.parse(stored) : [];
}

/**
 * saveFavorites(favoritesArray)
 * Writes the current favorites array to localStorage.
 */
function saveFavorites(favoritesArray) {
  localStorage.setItem(FAVORITES_KEY, JSON.stringify(favoritesArray));
}

/**
 * toggleFavorite(productId)
 * Adds the product to favorites if it is not already saved;
 * removes it if it is. Saves the updated array and refreshes
 * the favorites panel.
 */
function toggleFavorite(productId) {
  let favorites = loadFavorites();
  const index = favorites.indexOf(productId);

  if (index === -1) {
    favorites.push(productId);
  } else {
    favorites.splice(index, 1);
  }

  saveFavorites(favorites);
  updateFavoritesPanel();
  updateHeartButtons();
}

/**
 * updateHeartButtons()
 * Reads the current favorites from localStorage and updates
 * every heart button on the page to show filled or empty state.
 */
function updateHeartButtons() {
  const favorites = loadFavorites();
  const buttons = document.querySelectorAll(".fav-btn");

  buttons.forEach(function (btn) {
    const productId = btn.getAttribute("data-product-id");
    if (favorites.includes(productId)) {
      btn.textContent = "♥ Saved";
      btn.classList.add("fav-btn--active");
      btn.setAttribute("aria-pressed", "true");
    } else {
      btn.textContent = "♡ Save";
      btn.classList.remove("fav-btn--active");
      btn.setAttribute("aria-pressed", "false");
    }
  });
}

/**
 * updateFavoritesPanel()
 * Rebuilds the favorites summary panel that lists saved items.
 * Shows a placeholder message when no items are saved.
 */
function updateFavoritesPanel() {
  const panel = document.getElementById("favorites-panel");
  if (!panel) return;

  const favorites = loadFavorites();
  const list = document.getElementById("favorites-list");
  const emptyMsg = document.getElementById("favorites-empty");
  const countEl = document.getElementById("favorites-count");

  list.innerHTML = "";

  if (favorites.length === 0) {
    emptyMsg.style.display = "block";
    countEl.textContent = "0";
    return;
  }

  emptyMsg.style.display = "none";
  countEl.textContent = favorites.length;

  favorites.forEach(function (productId) {
    const product = products.find(function (p) { return p.id === productId; });
    if (!product) return;

    const li = document.createElement("li");
    li.textContent = product.name + " (" + product.category + ")";

    const removeBtn = document.createElement("button");
    removeBtn.textContent = "Remove";
    removeBtn.className = "fav-remove-btn";
    removeBtn.setAttribute("aria-label", "Remove " + product.name + " from favorites");
    removeBtn.addEventListener("click", function () {
      toggleFavorite(productId);
    });

    li.appendChild(removeBtn);
    list.appendChild(li);
  });
}

/**
 * initFavoritesFeature()
 * Sets up the favorites feature on products.html.
 * Attaches click listeners to all heart buttons and
 * renders the initial state from localStorage.
 */
function initFavoritesFeature() {
  const favButtons = document.querySelectorAll(".fav-btn");
  if (favButtons.length === 0) return; // not on products page

  favButtons.forEach(function (btn) {
    btn.addEventListener("click", function () {
      const productId = btn.getAttribute("data-product-id");
      toggleFavorite(productId);
    });
  });

  updateFavoritesPanel();
  updateHeartButtons();
}

/* ============================================================
   FEATURE 2 — CONTACT FORM VALIDATION
   Appears on contact.html
   ============================================================ */

/**
 * showError(fieldId, message)
 * Displays an error message directly below the specified field.
 * Creates the error element if it does not already exist.
 */
function showError(fieldId, message) {
  const field = document.getElementById(fieldId);
  if (!field) return;

  let errorEl = document.getElementById(fieldId + "-error");
  if (!errorEl) {
    errorEl = document.createElement("span");
    errorEl.id = fieldId + "-error";
    errorEl.className = "field-error";
    errorEl.setAttribute("role", "alert");
    field.parentNode.insertBefore(errorEl, field.nextSibling);
  }

  errorEl.textContent = message;
  field.classList.add("input-invalid");
  field.setAttribute("aria-describedby", fieldId + "-error");
}

/**
 * clearError(fieldId)
 * Removes the error message and invalid styling from a field.
 */
function clearError(fieldId) {
  const field = document.getElementById(fieldId);
  if (!field) return;

  const errorEl = document.getElementById(fieldId + "-error");
  if (errorEl) errorEl.textContent = "";

  field.classList.remove("input-invalid");
}

/**
 * validateName()
 * Checks that the name field is not empty and meets
 * the minimum length requirement.
 * Returns true if valid, false otherwise.
 */
function validateName() {
  const field = document.getElementById("name");
  if (!field) return true;
  const value = field.value.trim();

  if (value.length < validationRules.name.minLength) {
    showError("name", validationRules.name.errorMsg);
    return false;
  }
  clearError("name");
  return true;
}

/**
 * validateEmail()
 * Checks that the email field contains a properly formatted
 * email address using a regular expression pattern.
 * Returns true if valid, false otherwise.
 */
function validateEmail() {
  const field = document.getElementById("email");
  if (!field) return true;
  const value = field.value.trim();

  if (!validationRules.email.pattern.test(value)) {
    showError("email", validationRules.email.errorMsg);
    return false;
  }
  clearError("email");
  return true;
}

/**
 * validatePickupDate()
 * Checks that a pickup date has been selected and that it is
 * at least 24 hours in the future (48 hours for custom cakes).
 * Returns true if valid, false otherwise.
 */
function validatePickupDate() {
  const field = document.getElementById("pickup-date");
  const requestTypeField = document.getElementById("request-type");
  if (!field) return true;

  const selectedDate = new Date(field.value);
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const minHours = (requestTypeField && requestTypeField.value === "pre-order-cake") ? 48 : 24;
  const minDate = new Date(today.getTime() + minHours * 60 * 60 * 1000);

  if (!field.value || selectedDate < minDate) {
    const msg = minHours === 48
      ? "Custom cake orders require at least 48 hours advance notice."
      : validationRules.pickupDate.errorMsg;
    showError("pickup-date", msg);
    return false;
  }
  clearError("pickup-date");
  return true;
}

/**
 * validateRequestType()
 * Checks that the user has selected a request type from
 * the dropdown. Returns true if valid, false otherwise.
 */
function validateRequestType() {
  const field = document.getElementById("request-type");
  if (!field) return true;

  if (!field.value) {
    showError("request-type", validationRules.requestType.errorMsg);
    return false;
  }
  clearError("request-type");
  return true;
}

/**
 * validateItemDetails()
 * Checks that the item details textarea contains a meaningful
 * description of at least 10 characters.
 * Returns true if valid, false otherwise.
 */
function validateItemDetails() {
  const field = document.getElementById("item-details");
  if (!field) return true;
  const value = field.value.trim();

  if (value.length < validationRules.itemDetails.minLength) {
    showError("item-details", validationRules.itemDetails.errorMsg);
    return false;
  }
  clearError("item-details");
  return true;
}

/**
 * validateForm()
 * Runs all validation checks and returns true only if every
 * check passes. Focuses the first invalid field.
 */
function validateForm() {
  const checks = [
    validateName(),
    validateEmail(),
    validatePickupDate(),
    validateRequestType(),
    validateItemDetails()
  ];

  const allValid = checks.every(function (result) { return result === true; });

  if (!allValid) {
    const firstInvalid = document.querySelector(".input-invalid");
    if (firstInvalid) firstInvalid.focus();
  }

  return allValid;
}

/* ----------------------------------------------------------
   STORAGE: Save and restore contact name
   ---------------------------------------------------------- */

/**
 * saveContactName()
 * Saves the value of the name field to localStorage so it
 * can be pre-filled on the next visit.
 */
function saveContactName() {
  const nameField = document.getElementById("name");
  if (nameField && nameField.value.trim().length > 0) {
    localStorage.setItem(NAME_KEY, nameField.value.trim());
  }
}

/**
 * restoreContactName()
 * Reads the saved name from localStorage and pre-fills
 * the name field if a value exists.
 */
function restoreContactName() {
  const nameField = document.getElementById("name");
  if (!nameField) return;

  const savedName = localStorage.getItem(NAME_KEY);
  if (savedName) {
    nameField.value = savedName;
  }
}

/**
 * initContactForm()
 * Sets up form validation and storage on contact.html.
 * Attaches the submit handler and restores the saved name.
 */
function initContactForm() {
  const form = document.querySelector(".contact-form");
  if (!form) return;

  restoreContactName();

  // Live validation: clear errors as the user corrects fields
  document.getElementById("name")        && document.getElementById("name").addEventListener("input", validateName);
  document.getElementById("email")       && document.getElementById("email").addEventListener("input", validateEmail);
  document.getElementById("pickup-date") && document.getElementById("pickup-date").addEventListener("change", validatePickupDate);
  document.getElementById("request-type")&& document.getElementById("request-type").addEventListener("change", function () {
    validateRequestType();
    validatePickupDate(); // re-check date when type changes (cake vs other)
  });
  document.getElementById("item-details")&& document.getElementById("item-details").addEventListener("input", validateItemDetails);

  form.addEventListener("submit", function (event) {
    event.preventDefault();

    if (validateForm()) {
      saveContactName();
      // Show a success confirmation message
      showSuccessMessage();
    }
  });
}

/**
 * showSuccessMessage()
 * Replaces the form with a friendly confirmation message
 * after a valid submission.
 */
function showSuccessMessage() {
  const formSection = document.querySelector(".contact-form-section");
  if (!formSection) return;

  formSection.innerHTML =
    "<div class='form-success'>" +
    "<h2>Thank You!</h2>" +
    "<p>Your request has been received. We will get back to you within one business day.</p>" +
    "<p>We look forward to baking for you!</p>" +
    "</div>";
}

/* ============================================================
   INIT — Run the right feature based on the current page
   ============================================================ */
document.addEventListener("DOMContentLoaded", function () {
  initFavoritesFeature();
  initContactForm();
});
