// Favorites array
let favorites = [];

// Toggle favorite on or off
function toggleFavorite(productId) {
  productId = productId.trim();

  const btn = document.querySelector(`[data-product-id="${productId}"]`);
  if (!btn) {
    console.warn(`toggleFavorite: no button found for "${productId}"`);
    return;
  }

  const alreadySaved = favorites.some(item => item.id === productId);

  if (alreadySaved) {
    favorites = favorites.filter(item => item.id !== productId);
    btn.textContent = "☆ Save to Favorites";
  } else {
    // Read name and price directly from the card, not a separate data object
    const card      = btn.closest(".card-body");
    const name      = card?.querySelector("h3")?.textContent.trim()  ?? productId;
    const price     = card?.querySelector(".price")?.textContent.trim() ?? "";

    favorites.push({ id: productId, name, price });
    btn.textContent = "♥ Saved!";
  }

  saveFavorites();
  renderFavorites();
}

// Save favorites array to localStorage
function saveFavorites() {
  localStorage.setItem("bakeryFavorites", JSON.stringify(favorites));
}

// Display the favorites list on the page
function renderFavorites() {
  const list = document.getElementById("favorites-list");
  if (!list) return;

  list.innerHTML = "";

  if (favorites.length === 0) {
    const empty = document.createElement("p");
    empty.textContent = "No favorites saved yet.";
    list.appendChild(empty);
    return;
  }

  favorites.forEach(item => {
    const li = document.createElement("li");

    const nameSpan = document.createElement("span");
    nameSpan.textContent = `${item.name} — ${item.price}`;

    const removeBtn = document.createElement("button");
    removeBtn.textContent = "Remove";
    removeBtn.className = "remove-btn";
    removeBtn.setAttribute("aria-label", `Remove ${item.name} from favorites`);
    removeBtn.addEventListener("click", () => toggleFavorite(item.id));

    li.appendChild(nameSpan);
    li.appendChild(removeBtn);
    list.appendChild(li);
  });
}

// Load favorites from localStorage when the page opens
function loadFavorites() {
  try {
    const stored = localStorage.getItem("bakeryFavorites");
    if (stored) {
      favorites = JSON.parse(stored);
      renderFavorites();

      // Restore heart button states
      favorites.forEach(item => {
        const btn = document.querySelector(`[data-product-id="${item.id}"]`);
        if (btn) btn.textContent = "♥ Saved!";
      });
    }
  } catch (error) {
    console.error("Failed to load favorites from localStorage:", error);
    favorites = [];
  }
}

window.addEventListener("DOMContentLoaded", loadFavorites);
